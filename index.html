// WebGPU WGSL Fragment Shader: Physical Cook-Torrance BRDF
fn cookTorranceSpecular(N: vec3<f32>, V: vec3<f32>, L: vec3<f32>, roughness: f32) -> f32 {
  let H = normalize(V + L);
  let NdotH = max(dot(N, H), 0.001);
  let NdotV = max(dot(N, V), 0.001);
  let NdotL = max(dot(N, L), 0.001);
  let alpha = roughness * roughness;
  let alpha2 = alpha * alpha;
  // GGX / Trowbridge-Reitz NDF
  let denom = NdotH * NdotH * (alpha2 - 1.0) + 1.0;
  let D = alpha2 / (3.14159 * denom * denom);
  // Schlick-Smith Geometric Attenuation
  let k = (roughness + 1.0) * (roughness + 1.0) / 8.0;
  let G = (NdotV / (NdotV * (1.0 - k) + k)) *
          (NdotL / (NdotL * (1.0 - k) + k));
  return D * G / (4.0 * NdotV * NdotL);
}
// NeuralEngine.ts — ONNX Runtime Web Depth Inference
import * as ort from 'onnxruntime-web';

export async function initDepthEngine(): Promise<ort.InferenceSession> {
  ort.env.wasm.numThreads = Math.min(navigator.hardwareConcurrency, 4);
  ort.env.wasm.simd = true;
  ort.env.wasm.proxy = true;

  const session = await ort.InferenceSession.create(
    '/assets/depth_anything_v2_small.onnx',
    {
      executionProviders: ['webgpu', 'wasm'],
      graphOptimizationLevel: 'all',
      executionMode: 'parallel'
    }
  );
  return session;
}

export async function runDepthInference(
  session: ort.InferenceSession,
  pixels: Float32Array,
  w: number, h: number
): Promise<Float32Array> {
  const tensor = new ort.Tensor('float32', pixels, [1, 3, h, w]);
  const results = await session.run({ input: tensor });
  return results.output.data as Float32Array;
}
// WebGPUContext.ts — Device & Render Pipeline Bootstrap
export interface GPUContext {
  device: GPUDevice;
  context: GPUCanvasContext;
  format: GPUTextureFormat;
}

export async function initWebGPU(
  canvas: HTMLCanvasElement
): Promise<GPUContext> {
  if (!navigator.gpu) throw new Error('WebGPU not supported');

  const adapter = await navigator.gpu.requestAdapter({
    powerPreference: 'high-performance'
  });
  const device = await adapter?.requestDevice();
  const context = canvas.getContext('webgpu');
  const format = navigator.gpu.getPreferredCanvasFormat();

  context.configure({ device, format, alphaMode: 'premultiplied' });
  return { device, context, format };
}

Architecture & Principles

Engineered for Local Performance

Orlume runs machine learning models and 3D graphics entirely inside your browser. No cloud servers, no queues.
STAGE 01

Local Ingestion

Images decode directly into browser GPU texture memory with zero network requests.

STAGE 02

Neural Inference

Depth Anything V2 executes via ONNX Runtime Web with 128-bit WebAssembly SIMD multi-threading.

STAGE 03

Shader Lighting

WebGPU/WebGL2 fragment shaders compute normal vectors and physical specular highlights at 60 FPS.

STAGE 04

Lossless Export

Full-resolution PNG, WebP, or JPEG files are generated directly on your machine without cloud compression.

Private by Design

Your photos never leave your machine. No telemetry, no background analytics, and no training on user images.

Sub-16ms Latency

Direct GPU computation eliminates network round-trips, giving you instant, reversible visual feedback as you adjust lights.
